```typescript
export class PM5Service extends EventTarget {
  private device: BluetoothDevice | null = null;
  private reconnectAttempts = 0;
  private readonly MAX_RECONNECT_ATTEMPTS = 3;
  private readonly RECONNECT_INTERVAL_MS = 30000; // BLE-05: 30s interval
  private reconnectTimer: ReturnType<typeof setTimeout> | null = null;

  // MUST be called from a user gesture — BLE-01
  async connect(): Promise<void> { ... }

  private async connectGatt(): Promise<void> { ... }

  // Subscribes to all 3 PM5 GATT characteristics — BLE-02
  private async subscribeCharacteristics(service: BluetoothRemoteGATTService): Promise<void> { ... }

  // Parses bytes and dispatches 'stroke_reading' CustomEvent — BLE-02
  private handleRowingStatus = (event: Event): void => { ... }
  private handleExtraStatus1 = (event: Event): void => { ... }
  private handleExtraStatus2 = (event: Event): void => { ... }

  // BLE-04: detect disconnect, dispatch 'disconnected', start reconnect loop
  private onDisconnected = (): void => {
    this.dispatchEvent(new CustomEvent('disconnected'));
    this.scheduleReconnect();
  };

  // BLE-05: attempt reconnect every 30s, max 3 attempts
  private scheduleReconnect(): void {
    if (this.reconnectAttempts >= this.MAX_RECONNECT_ATTEMPTS) {
      this.dispatchEvent(new CustomEvent('reconnect_failed'));
      return;
    }
    this.reconnectTimer = setTimeout(async () => {
      this.reconnectAttempts++;
      this.dispatchEvent(new CustomEvent('reconnecting', { detail: { attempt: this.reconnectAttempts } }));
      try {
        await this.connectGatt();
        this.reconnectAttempts = 0; // reset on success
        this.dispatchEvent(new CustomEvent('connected'));
      } catch {
        this.scheduleReconnect();
      }
    }, this.RECONNECT_INTERVAL_MS);
  }

  disconnect(): void {
    if (this.reconnectTimer) clearTimeout(this.reconnectTimer);
    this.device?.gatt?.disconnect();
  }
}
```

```typescript
export class WakeLockManager {
  private sentinel: WakeLockSentinel | null = null;

  // BLE-03: acquire screen wake lock
  async acquire(): Promise<void> {
    if (!('wakeLock' in navigator)) {
      console.warn('Screen Wake Lock API not supported');
      return;
    }
    try {
      this.sentinel = await navigator.wakeLock.request('screen');
    } catch (err) {
      console.warn('Wake Lock not acquired:', err);
    }
  }

  release(): void {
    this.sentinel?.release();
    this.sentinel = null;
  }

  // BLE-03: re-acquire on visibility restore — call once at session start
  bindVisibilityChange(): void {
    document.addEventListener('visibilitychange', async () => {
      if (document.visibilityState === 'visible' && this.sentinel === null) {
        await this.acquire();
      }
    });
  }

  get isHeld(): boolean {
    return this.sentinel !== null;
  }
}
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Tampering | pm5-parser.ts DataView bytes | mitigate | validateStrokeReading() throws RangeError on out-of-range values before data reaches session state. Corrupt GATT bytes cannot produce impossible metric values (sub-1-minute split, negative watts, 70spm). |
| T-02-02 | Denial of Service | BLE reconnect loop | mitigate | MAX_RECONNECT_ATTEMPTS=3 caps reconnect attempts. reconnect_failed event triggers UI fallback mode. No infinite retry that could drain battery. |
| T-02-03 | Information Disclosure | BLE device name/address in session state | accept | PM5 device name (e.g. "PM5 12345") is low-sensitivity hardware identifier. Stored in memory only, not transmitted to server. Acceptable for v1. |
| T-02-04 | Spoofing | BLE device filter | mitigate | `requestDevice` filtered by `PM5_SERVICE_UUID` — only devices advertising the Concept2 PM5 GATT service can be selected. Prevents random BLE devices from being used as a PM5. |
</threat_model>
